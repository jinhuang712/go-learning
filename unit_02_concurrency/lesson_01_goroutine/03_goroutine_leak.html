1.3 Goroutine 泄露初体验

3.1 什么是 Goroutine 泄露

Goroutine 泄露(leak)指一个 Goroutine 启动后,因为某种原因永远无法结束, 从而它的栈、引用的对象、文件描述符等资源都无法回收。

关键认知 Go GC 不会回收阻塞中的 Goroutine。只要它还在等 channel / 锁 / 网络, 它对应的栈和它栈上的引用就是「GC roots」,所有这些对象都没法释放。 Goroutine 泄露 = Go 程序内存增长的头号元凶。

3.2 常见的 4 种泄露场景

  1. Channel 阻塞: Goroutine 在等一个永远不会有数据的 channel
  2. 锁未释放: 持有锁的 Goroutine 崩溃或忘记释放,其他 Goroutine 永远等
  3. 循环引用: 多个 Goroutine 之间互等(死锁,但 Go runtime 不一定检测得到)
  4. 缺少退出机制: 无限循环 + 没有退出信号

3.5 错误写法:Channel 接收导致泄露

package main

import (
    "fmt"
    "runtime"
    "time"
)

func main() {
    fmt.Printf("初始 Goroutine 数量: %d\n", runtime.NumGoroutine())

    // 启动 100 个会泄露的 Goroutine
    for i := 0; i < 100; i++ {
        ch := make(chan int)
        go func() {
            <-ch // 永远等待,没有人发送
        }()
    }

    time.Sleep(100 * time.Millisecond)
    fmt.Printf("泄露后 Goroutine 数量: %d\n", runtime.NumGoroutine())
    // 输出: 初始 1, 泄露后 101 — 多出来的 100 个永远不会消失
}

3.6 正确写法

package main

import (
    "fmt"
    "runtime"
    "time"
)

func main() {
    fmt.Printf("初始 Goroutine 数量: %d\n", runtime.NumGoroutine())

    // 方案 1: 用带缓冲 channel,保证 send 不阻塞
    for i := 0; i < 10; i++ {
        ch := make(chan int, 1)
        go func() {
            <-ch
        }()
        ch <- 1 // 主动发送,worker 收到后就能退出
    }

    // 方案 2: 用 done channel 显式通知退出
    done := make(chan struct{})
    go func() {
        select {
        case <-done:
            return
        }
    }()
    close(done) // close 后,所有 receiver 都立即收到零值

    time.Sleep(100 * time.Millisecond)
    fmt.Printf("修复后 Goroutine 数量: %d\n", runtime.NumGoroutine())
}

3.7 生产环境三大坑点

坑点 1: 忘记接收者,send 阻塞 
// 错误写法
go func() {
    result := doSomething()
    ch <- result // 如果没有人接收,会阻塞 → 泄露
}()

// 正确写法 1: 带缓冲 channel
ch := make(chan Result, 1)
go func() {
    result := doSomething()
    ch <- result // 缓冲区有位,不阻塞
}()

// 正确写法 2: 用 select + default 实现非阻塞发送
go func() {
    result := doSomething()
    select {
    case ch <- result:
    default: // 没人收就丢弃,避免泄露
    }
}()
坑点 2: 忘记 close,range 永远阻塞 
// 错误写法
ch := make(chan int)
go func() {
    for i := 0; i < 10; i++ {
        ch <- i
    }
    // 忘记 close(ch)
}()

for val := range ch { // ⚠ 永远阻塞 — 这是泄露
    fmt.Println(val)
}

// 正确写法: defer close 让所有 range 干净退出
go func() {
    defer close(ch)
    for i := 0; i < 10; i++ {
        ch <- i
    }
}()
坑点 3: HTTP Handler 里裸启 Goroutine 
// 错误写法: 后台 Goroutine 没有超时控制,请求结束后还在跑
http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
    go func() {
        time.Sleep(time.Hour) // 每来一个请求就泄露一个 G
    }()
})

// 正确写法: 用 Context 控制生命周期
http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
    ctx, cancel := context.WithTimeout(r.Context(), 5*time.Second)
    defer cancel()

    go func() {
        select {
        case <-ctx.Done():
            return // 请求结束/超时,Goroutine 干净退出
        case <-time.After(time.Hour):
            // 业务超时
        }
    }()
})
诊断技巧 生产环境怀疑泄露时,挂上 net/http/pprof,访问 /debug/pprof/goroutine?debug=2, 就能看到所有当前 Goroutine 的调用栈。哪一行阻塞、阻塞多久,一目了然。

📝 练习题

练习 3.1: 找出下面代码中的 所有 问题(不止一个)并修复:

package main

import (
    "fmt"
    "time"
)

func processTask(taskID int) string {
    time.Sleep(100 * time.Millisecond)
    return fmt.Sprintf("Result of task %d", taskID)
}

func main() {
    results := make([]string, 0)

    for i := 0; i < 10; i++ {
        go func(taskID int) {
            result := processTask(taskID)
            results = append(results, result) // 这里有问题!
        }(i)
    }

    time.Sleep(1 * time.Second)
    fmt.Printf("Got %d results\n", len(results))
}

提示: 至少有两个问题 — (1) 共享 slice 的 data race(append 并发不安全), (2) time.Sleep 不是同步原语,生产代码不能这样等。
